#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void main(int argc, char *argv[])
{
  char *name = malloc(50);  
  int age = 10;
  strcpy(name, "Hiran");
  
  if (argc>=3) age = atoi(argv[2]);
  if (argc>=2) strcpy(name, argv[1]);
  
  printf("Hello %s, you are %d years old.", name, age);
}
