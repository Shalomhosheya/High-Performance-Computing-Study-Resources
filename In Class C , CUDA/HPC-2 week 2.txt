#include <stdio.h>

int main() {
    float height, weight, bmi;

    printf("Enter height (m): ");
    scanf("%f", &height);

    printf("Enter weight (kg): ");
    scanf("%f", &weight);

    bmi = weight / (height * height);

    printf("BMI = %.2f\n", bmi);

    return 0;
}
