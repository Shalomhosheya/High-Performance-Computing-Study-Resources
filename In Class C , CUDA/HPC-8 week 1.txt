%%writefile hello.cu

#include <stdio.h>

__global__ void cuda_hello() {
    printf("\n Hello World from GPU Block ID %d Threads ID %d\n", blockIdx.x, threadIdx.x);
}

int main() {
    printf("CPU Hello World!\n");

    cuda_hello<<<2, 3>>>();

    cudaDeviceSynchronize();

    return 0;
}
