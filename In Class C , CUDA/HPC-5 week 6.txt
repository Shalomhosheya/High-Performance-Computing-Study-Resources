#include <stdio.h>

int main()
{
    const int SIZE = 3;
    int i, arr[SIZE], sum = 0;

    printf("Enter Numbers: ");

    for (i = 0; i < SIZE; i++)
        scanf("%d", &arr[i]);

    for (i = 0; i < SIZE; i++)
        sum = sum + arr[i];

    printf("Sum is %d", sum);

    return 0;
}
