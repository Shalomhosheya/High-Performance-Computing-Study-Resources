#include <omp.h>
#include <stdio.h>

int main()
{
    int nstart = 1;
    int nfinish = 20000;

    printf("Prime numbers between %d and %d are:\n", nstart, nfinish);

    #pragma omp parallel for
    for (int i = nstart; i <= nfinish; i++)
    {
        int c;

        for (c = 2; c <= i - 1; c++)
        {
            if (i % c == 0)
                break;
        }

        if (c == i)
        {
            printf("%d\n", i);
        }
    }

    return 0;
}
