%%writefile MatrixMultiplication.cu

#include <stdio.h>
#include <stdlib.h>
#include <cuda_runtime.h>

#define AR 3
#define AC 4
#define BR 4
#define BC 2

void random_ints(int *a, int n)
{
    for (int i = 0; i < n; i++)
    {
        a[i] = rand() % 10;
    }
}

__global__ void add(int *a, int *b, int *c,
                    int mASize, int mBSize, int mCSize)
{
    int row = blockIdx.x;
    int col = threadIdx.x;

    if (row < AR && col < BC)
    {
        int sum = 0;

        for (int k = 0; k < AC; k++)
        {
            sum += a[row * AC + k] * b[k * BC + col];
        }

        c[row * BC + col] = sum;
    }
}

int main(void)
{
    // Host copies of a, b, c
    int *a, *b, *c;

    // Device copies of a, b, c
    int *d_a, *d_b, *d_c;

    int mASize = AR * AC * sizeof(int);
    int mBSize = BR * BC * sizeof(int);
    int mCSize = AR * BC * sizeof(int);

    // Allocate space for device copies
    cudaMalloc((void**)&d_a, mASize);
    cudaMalloc((void**)&d_b, mBSize);
    cudaMalloc((void**)&d_c, mCSize);

    // Allocate space for host copies
    a = (int *)malloc(mASize);
    b = (int *)malloc(mBSize);
    c = (int *)malloc(mCSize);

    // Generate random values
    random_ints(a, AR * AC);
    random_ints(b, BR * BC);

    // Copy inputs to device
    cudaMemcpy(d_a, a, mASize, cudaMemcpyHostToDevice);
    cudaMemcpy(d_b, b, mBSize, cudaMemcpyHostToDevice);

    // Launch kernel
    add<<<AR, BC>>>(d_a, d_b, d_c, mASize, mBSize, mCSize);

    // Wait for GPU
    cudaDeviceSynchronize();

    // Copy result back to host
    cudaMemcpy(c, d_c, mCSize, cudaMemcpyDeviceToHost);

    // Print Matrix A
    printf("Matrix A:\n");

    for (int i = 0; i < AR; i++)
    {
        for (int j = 0; j < AC; j++)
        {
            printf("%d ", a[i * AC + j]);
        }
        printf("\n");
    }

    // Print Matrix B
    printf("\nMatrix B:\n");

    for (int i = 0; i < BR; i++)
    {
        for (int j = 0; j < BC; j++)
        {
            printf("%d ", b[i * BC + j]);
        }
        printf("\n");
    }

    // Print Matrix C
    printf("\nMatrix C = A x B:\n");

    for (int i = 0; i < AR; i++)
    {
        for (int j = 0; j < BC; j++)
        {
            printf("%d ", c[i * BC + j]);
        }
        printf("\n");
    }

    // Cleanup
    free(a);
    free(b);
    free(c);

    cudaFree(d_a);
    cudaFree(d_b);
    cudaFree(d_c);

    return 0;
}
