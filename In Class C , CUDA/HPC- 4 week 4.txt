#include <stdio.h>

int main()
{
    int n, isDive3 = 0, isDive5 = 0, isDive7 = 0;

    printf("Enter No: ");
    scanf("%d", &n);

    for (int i = 1; i <= n; i++)
    {
        isDive3 = ! (i % 3);
        isDive5 = ! (i % 5);
        isDive7 = ! (i % 7);

        if (isDive3 == 0 || isDive5 == 0 || isDive7 == 0)
		printf(" \n ");
        {
            if (isDive3 == 0)
                printf(" Bish ");

            if (isDive5 == 0)
                printf(" Bash ");

            if (isDive7 == 0)
                printf(" Bosh ");
        }

        printf("\n%5d", i);
    }

    return 0;
}
