#include <stdio.h>

int main()
{
    const int SIZE = 3;
    int i, arr[SIZE], max;

    printf("Enter Numbers: ");

    for (i = 0; i < SIZE; i++)
        scanf("%d", &arr[i]);

    max = arr[0];

    for (i = 1; i < SIZE; i++ )
    {
        if (arr[i] > max)
            max = arr[i];
    }

    printf("Maximum is %d", max);

    return 0;
}
