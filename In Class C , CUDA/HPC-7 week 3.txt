#include <stdio.h>
#include <stdlib.h>

struct args
{
    int **mA;
    int **mB;
    int **mC;
    int rA, cA, rB, cB; 
};

void MatrixMatrixMultiplication(struct args * arg);

int main(){
   
   int **mA;
   int rA = 3, cA = 4, rB = 4, cB = 3, rC, cC, i;
   rC = rA;
   cC = cB;

    mA = (int**)malloc(rA*sizeof(int*));
    for (i=0;i<rA;i++)
        mA[i]= (int*)malloc(cA*sizeof(int));
 
   int **mB;
 
    mB = (int**)malloc(rB*sizeof(int*));
    for (i=0;i<rB;i++)
        mB[i]= (int*)malloc(cB*sizeof(int));
  
    int **mC;
 
    mC = (int**)malloc(rC*sizeof(int*));
    for (i=0;i<rC;i++)
        mC[i]= (int*)malloc(cC*sizeof(int));  
        
    int r=0, c=0;        
    FILE * fileIn, * fileOut;
    char str[500];
      
   if (fileIn = fopen("matrix.txt", "r")){
      printf("File opened successfully in read mode");
    
      printf("\n\n");
         
      while(fscanf(fileIn,"%s", str)!=EOF)
      {
         printf("%s", str);

         if (r<rA)
         {
             mA[r][c] = atoi(str);
    
             c++;
             c%=cA;
         } 
         else
         {
             mB[r-rA][c] = atoi(str);
    
             c++;
             c%=cB;         
         }
         
         if (c==0)
            r++;

      }
   }
   else
   {
        printf("The file is not present! cannot create a new file using r mode");
   }    
   
   printf("\n\n");
   
   for(r=0; r<rA; r++)
   {
        for(c=0; c<cA; c++)
        {
            printf("%d\t", mA[r][c]);
        }
        printf("\n");
   }

   for(r=0; r<rB; r++)
   {
        for(c=0; c<cB; c++)
        {
            printf("%d\t", mB[r][c]);
        }
        printf("\n");
   } 
   
   struct args arg;
   arg.mA = mA;
   arg.mB = mB;
   arg.mC = mC;
   arg.rA = rA;
   arg.cA = cA;
   arg.rB = rB;
   arg.cB = cB;
   
   MatrixMatrixMultiplication(&arg);
 
   fileOut = fopen("matrixOut.txt", "w");
   for(r=0; r<rC; r++)
   {
        for(c=0; c<cC; c++)
        {
            printf("%d\t", mC[r][c]);
            fprintf(fileOut,"%d ", mC[r][c]);
        }
        printf("\n");
        fprintf(fileOut,"\n");
   }
   
   
   fclose(fileIn);
   fclose(fileOut);
   
   return 0;
}

void MatrixMatrixMultiplication(struct args * arg)
{
    int **matrixA = arg->mA;
    int **matrixB = arg->mB; 
    int **matrixC = arg->mC;
    int rA = arg->rA;
    int cA = arg->cA; 
    int rB = arg->rB; 
    int cB = arg->cB;
    
    int i, j, k, tot;
    
    for(k=0; k<rA; k++)
    {
        for(j=0; j<cB; j++)
        {  
            tot = 0;
            for(i=0; i<cA; i++)
            {   
                tot += matrixA[k][i] * matrixB[i][j];
            }
            matrixC[k][j] = tot;
        }
    }
}
